PROJECT 3: MULTI-UNIT CONVERTER
-----------------------------------------------------------

OVERVIEW:
Create a conversion tool that handles multiple types of unit conversions.
Your converter should support at least 3 conversion categories.

REQUIREMENTS:
✓ Welcome message and menu of conversion types
✓ At least 3 conversion categories (temperature, distance, weight, etc.)
✓ Handle user input for values and units
✓ Perform accurate calculations
✓ Display results in a clean format
✓ Option to perform another conversion

STARTER CODE:

print("=" * 50)
print("UNIVERSAL UNIT CONVERTER")
print("=" * 50)
print("\nWelcome! This tool can convert between different units.\n")

print("Available Conversions:")
print("1. Temperature (Celsius/Fahrenheit)")
print("2. Distance (Miles/Kilometers)")
print("3. Weight (Pounds/Kilograms)")

choice = input("\nSelect a conversion type (1/2/3): ")

if choice == "1":
    # TEMPERATURE CONVERSION
    print("\n--- Temperature Conversion ---")
    print("A. Celsius to Fahrenheit")
    print("B. Fahrenheit to Celsius")

    temp_choice = input("Choose conversion (A/B): ").upper()

    if temp_choice == "A":
        celsius = float(input("Enter temperature in Celsius: "))
        fahrenheit = (celsius * 9/5) + 32
        print(f"\n{celsius}°C = {fahrenheit}°F")

    elif temp_choice == "B":
        fahrenheit = float(input("Enter temperature in Fahrenheit: "))
        celsius = (fahrenheit - 32) * 5/9
        print(f"\n{fahrenheit}°F = {celsius}°C")

    else:
        print("Invalid choice!")

elif choice == "2":
    # DISTANCE CONVERSION
    print("\n--- Distance Conversion ---")
    print("A. Miles to Kilometers")
    print("B. Kilometers to Miles")

    dist_choice = input("Choose conversion (A/B): ").upper()

    if dist_choice == "A":
        miles = float(input("Enter distance in miles: "))
        kilometers = miles * 1.60934
        print(f"\n{miles} miles = {kilometers:.2f} kilometers")

    elif dist_choice == "B":
        kilometers = float(input("Enter distance in kilometers: "))
        miles = kilometers / 1.60934
        print(f"\n{kilometers} km = {miles:.2f} miles")

    else:
        print("Invalid choice!")

elif choice == "3":
    # WEIGHT CONVERSION
    print("\n--- Weight Conversion ---")
    print("A. Pounds to Kilograms")
    print("B. Kilograms to Pounds")

    weight_choice = input("Choose conversion (A/B): ").upper()

    if weight_choice == "A":
        pounds = float(input("Enter weight in pounds: "))
        kilograms = pounds * 0.453592
        print(f"\n{pounds} lbs = {kilograms:.2f} kg")

    elif weight_choice == "B":
        kilograms = float(input("Enter weight in kilograms: "))
        pounds = kilograms / 0.453592
        print(f"\n{kilograms} kg = {pounds:.2f} lbs")

    else:
        print("Invalid choice!")

else:
    print("\nInvalid selection!")

print("\nThank you for using the converter!")

CONVERSION FORMULAS:
Temperature:
- Celsius to Fahrenheit: (C × 9/5) + 32
- Fahrenheit to Celsius: (F - 32) × 5/9

Distance:
- Miles to Kilometers: miles × 1.60934
- Kilometers to Miles: km ÷ 1.60934

Weight:
- Pounds to Kilograms: lbs × 0.453592
- Kilograms to Pounds: kg ÷ 0.453592

TIPS FOR SUCCESS:
- Use float() for decimal inputs
- Round results with :.2f for 2 decimal places
- Test with known conversions to verify accuracy
- Add input validation where possible
- Make menus clear and easy to navigate

CHALLENGE ADDITIONS:

- Create a "reverse" option to go back to main menu
- Add input validation (reject negative temps in Kelvin, etc.)
- Include fun facts about units after conversion
- Support multiple unit chains (feet -> inches -> centimeters)

# Main loop for multiple conversions
continue_converting = "yes"

while continue_converting.lower() == "yes":
    print("\nAvailable Conversions:")
    print("1. Temperature (Celsius/Fahrenheit/Kelvin)")
    print("2. Distance (Miles/Kilometers/Meters)")
    print("3. Weight (Pounds/Kilograms/Grams)")
    print("4. Time (Hours/Minutes/Seconds)")
    print("5. Volume (Gallons/Liters/Milliliters)")

    choice = input("\nSelect a conversion type (1/2/3/4/5): ")

    if choice == "1":
        # TEMPERATURE CONVERSION
        print("\n" + "=" * 50)
        print("TEMPERATURE CONVERSION")
        print("=" * 50)
        print("A. Celsius to Fahrenheit")
        print("B. Fahrenheit to Celsius")
        print("C. Celsius to Kelvin")
        print("D. Kelvin to Celsius")

        temp_choice = input("\nChoose conversion (A/B/C/D): ").upper()

        if temp_choice == "A":
            celsius = float(input("Enter temperature in Celsius: "))
            fahrenheit = (celsius * 9/5) + 32
            print(f"\n{celsius}°C = {fahrenheit:.2f}°F")

        elif temp_choice == "B":
            fahrenheit = float(input("Enter temperature in Fahrenheit: "))
            celsius = (fahrenheit - 32) * 5/9
            print(f"\n{fahrenheit}°F = {celsius:.2f}°C")

        elif temp_choice == "C":
            celsius = float(input("Enter temperature in Celsius: "))
            kelvin = celsius + 273.15
            print(f"\n{celsius}°C = {kelvin:.2f}K")

        elif temp_choice == "D":
            kelvin = float(input("Enter temperature in Kelvin: "))
            celsius = kelvin - 273.15
            print(f"\n{kelvin}K = {celsius:.2f}°C")

        else:
            print("\nInvalid choice!")

    elif choice == "2":
        # DISTANCE CONVERSION
        print("\n" + "=" * 50)
        print("DISTANCE CONVERSION")
        print("=" * 50)
        print("A. Miles to Kilometers")
        print("B. Kilometers to Miles")
        print("C. Meters to Feet")
        print("D. Feet to Meters")

        dist_choice = input("\nChoose conversion (A/B/C/D): ").upper()

        if dist_choice == "A":
            miles = float(input("Enter distance in miles: "))
            kilometers = miles * 1.60934
            print(f"\n{miles} miles = {kilometers:.2f} kilometers")

        elif dist_choice == "B":
            kilometers = float(input("Enter distance in kilometers: "))
            miles = kilometers / 1.60934
            print(f"\n{kilometers} km = {miles:.2f} miles")

        elif dist_choice == "C":
            meters = float(input("Enter distance in meters: "))
            feet = meters * 3.28084
            print(f"\n{meters} meters = {feet:.2f} feet")

        elif dist_choice == "D":
            feet = float(input("Enter distance in feet: "))
            meters = feet / 3.28084
            print(f"\n{feet} feet = {meters:.2f} meters")

        else:
            print("\nInvalid choice!")

    elif choice == "3":
        # WEIGHT CONVERSION
        print("\n" + "=" * 50)
        print("WEIGHT CONVERSION")
        print("=" * 50)
        print("A. Pounds to Kilograms")
        print("B. Kilograms to Pounds")
        print("C. Kilograms to Grams")
        print("D. Grams to Kilograms")

        weight_choice = input("\nChoose conversion (A/B/C/D): ").upper()

        if weight_choice == "A":
            pounds = float(input("Enter weight in pounds: "))
            kilograms = pounds * 0.453592
            print(f"\n{pounds} lbs = {kilograms:.2f} kg")

        elif weight_choice == "B":
            kilograms = float(input("Enter weight in kilograms: "))
            pounds = kilograms / 0.453592
            print(f"\n{kilograms} kg = {pounds:.2f} lbs")

        elif weight_choice == "C":
            kilograms = float(input("Enter weight in kilograms: "))
            grams = kilograms * 1000
            print(f"\n{kilograms} kg = {grams:.2f} g")

        elif weight_choice == "D":
            grams = float(input("Enter weight in grams: "))
            kilograms = grams / 1000
            print(f"\n{grams} g = {kilograms:.2f} kg")

        else:
            print("\nInvalid choice!")

    elif choice == "4":
        # TIME CONVERSION
        print("\n" + "=" * 50)
        print("TIME CONVERSION")

        print("D. Seconds to Minutes")

            hours = float(input("Enter time in hours: "))
            minutes = hours * 60
            print(f"\n{hours} hours = {minutes:.2f} minutes")

        elif time_choice == "B":
            minutes = float(input("Enter time in minutes: "))
            hours = minutes / 60
            print(f"\n{minutes} minutes = {hours:.2f} hours")

        elif time_choice == "C":
            minutes = float(input("Enter time in minutes: "))
            seconds = minutes * 60
            print(f"\n{minutes} minutes = {seconds:.2f} seconds")
